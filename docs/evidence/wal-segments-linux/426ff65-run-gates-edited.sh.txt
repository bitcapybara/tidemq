#!/usr/bin/env bash
set -euo pipefail
export PATH=/tools:/tools/node/bin:/usr/local/cargo/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
export GIT_OPTIONAL_LOCKS=0
features=tidemq-sdk/metrics,tidemq-sdk/opentelemetry,tidemq-protocol/codegen
step() {
  local name=$1 result started ended
  shift
  started=$(date -u +%Y-%m-%dT%H:%M:%SZ)
  printf 'Starting %s at %s\n' "$name" "$started"
  set +e
  "$@" > "/stage/logs/$name.log" 2>&1
  result=$?
  set -e
  ended=$(date -u +%Y-%m-%dT%H:%M:%SZ)
  printf '%s\n' "$result" > "/stage/logs/$name.exit"
  printf '%s\n%s\n' "$started" "$ended" > "/stage/logs/$name.time"
  printf 'Finished %s with exit %s at %s\n' "$name" "$result" "$ended"
  if [[ $result -ne 0 ]]; then
    tail -n 60 "/stage/logs/$name.log"
    exit "$result"
  fi
}
[[ $(git rev-parse HEAD) == 426ff6598c7b6a4337cc07ca79ac70d51d12b413 ]]
[[ -z $(git status --porcelain --untracked-files=no) ]]
step versions bash -c 'rustc --version --verbose; cargo --version; cargo clippy --version; rustfmt --version; cargo nextest --version'
step fmt cargo fmt --all -- --check
step clear-workspace cargo clean --workspace --locked
step clippy cargo clippy --workspace --all-targets --no-deps --locked --features "$features" -- -D warnings
step format-generated bash -c 'rustfmt --edition 2024 /work/crates/tidemq-protocol/src/generated/*.rs'
step generated-diff diff -qr /frozen-generated /work/crates/tidemq-protocol/src/generated
step production-audit cargo clippy --workspace --lib --bins --no-deps --locked --features "$features" -- -D warnings -D clippy::unwrap_used -D clippy::expect_used -D clippy::panic -D clippy::unimplemented -D clippy::todo -D clippy::unreachable -D clippy::indexing_slicing
step nextest cargo nextest run --workspace --all-targets --locked --features "$features" --profile ci
cp /stage/target/nextest/ci/junit.xml /stage/logs/workspace-junit.xml
step doctests cargo test --workspace --doc --locked --features "$features"
step sdk-clippy cargo clippy -p tidemq-sdk --all-targets --no-deps --locked -- -D warnings
step sdk-nextest cargo nextest run -p tidemq-sdk --all-targets --locked --profile ci
cp /stage/target/nextest/ci/junit.xml /stage/logs/sdk-junit.xml
step validate-config cargo run --locked --bin tidemq -- validate-config --config config/local.toml
step allocator-probe cargo run --locked --bin tidemq -- allocator-probe --blocks 2
step trust-boundaries bash -c 'node --test scripts/coverage/*.test.mjs scripts/performance/*.test.mjs'
step rustfs cargo nextest run -p tidemq --locked --test s3_rustfs --test lifecycle_compaction --test lifecycle_persistence --test runtime_queue --lib --run-ignored only --profile ci -E 'not test(wal::segmented::fault_tests::crash_boundary_child)'
printf 'All pre-release gates completed; release paused by coordinator\n'; exit 0             
mkdir -p /stage/runtime/bin
install -m 0555 /stage/target/release/tidemq /stage/runtime/bin/tidemq
install -m 0555 /stage/target/release/examples/deploy_validation /stage/runtime/bin/deploy_validation
sha256sum /stage/runtime/bin/tidemq /stage/runtime/bin/deploy_validation > /stage/logs/binaries.sha256
[[ -z $(git status --porcelain --untracked-files=no) ]]
printf 'All normal Linux gates completed successfully\n'
