#!/usr/bin/env python3
"""Advise clean pages from approved private targets; never change file contents."""

import argparse
import datetime
import hashlib
import json
import os
from pathlib import Path
import re
import stat
import subprocess
import sys


def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def inspect_container(container_id, expected):
    result = subprocess.run(
        ["docker", "inspect", container_id], capture_output=True, text=True,
        check=True, timeout=10,
    )
    item = json.loads(result.stdout)[0]
    limits = {name: item["HostConfig"].get(name) for name in expected["limits"]}
    if (item["Id"] != container_id or item["Name"] != expected["name"]
            or item["Image"] != expected["image_id"]
            or not item["State"]["Running"] or limits != expected["limits"]):
        raise RuntimeError("The approved private container identity or limits changed")
    pid = item["State"]["Pid"]
    lines = Path(f"/proc/{pid}/cgroup").read_text().splitlines()
    paths = [line.split(":", 2)[2] for line in lines if line.startswith("0::")]
    if len(paths) != 1 or container_id not in paths[0]:
        raise RuntimeError("The private cgroup does not contain the exact container ID")
    cgroup = Path("/sys/fs/cgroup") / paths[0].lstrip("/")
    if cgroup.resolve(strict=True) != cgroup:
        raise RuntimeError("The private cgroup path is not canonical")
    return {"container_id": container_id, "name": item["Name"], "image_id": item["Image"],
            "pid": pid, "cgroup_path": str(cgroup), "limits": limits}


def resources(identity):
    root = Path(identity["cgroup_path"])
    names = ["memory.current", "memory.max", "memory.peak", "memory.events",
             "memory.stat", "memory.swap.current", "memory.swap.max", "pids.current"]
    return {name: (root / name).read_text().strip() for name in names}


def advise_root(root):
    path = Path(root)
    if not path.is_absolute() or path.resolve(strict=True) != path:
        raise RuntimeError("The approved private target root is not canonical")
    counts = {"root": str(path), "visited": 0, "advised": 0, "advised_file_bytes": 0,
              "skipped_symlink": 0, "skipped_hardlink": 0,
              "skipped_nonregular": 0, "skipped_other_device": 0,
              "error_count": 0, "errors": []}
    flags = os.O_RDONLY | os.O_CLOEXEC | os.O_NOFOLLOW
    root_fd = os.open(path, flags | os.O_DIRECTORY)
    try:
        device = os.fstat(root_fd).st_dev
        for relative, directories, files, dir_fd in os.fwalk(
                ".", topdown=True, follow_symlinks=False, dir_fd=root_fd):
            for name in list(directories):
                info = os.stat(name, dir_fd=dir_fd, follow_symlinks=False)
                if stat.S_ISLNK(info.st_mode) or info.st_dev != device:
                    directories.remove(name)
                    key = "skipped_symlink" if stat.S_ISLNK(info.st_mode) else "skipped_other_device"
                    counts[key] += 1
            for name in files:
                counts["visited"] += 1
                try:
                    info = os.stat(name, dir_fd=dir_fd, follow_symlinks=False)
                    if stat.S_ISLNK(info.st_mode):
                        counts["skipped_symlink"] += 1
                        continue
                    if not stat.S_ISREG(info.st_mode):
                        counts["skipped_nonregular"] += 1
                        continue
                    fd = os.open(name, flags, dir_fd=dir_fd)
                    try:
                        opened = os.fstat(fd)
                        if not stat.S_ISREG(opened.st_mode):
                            counts["skipped_nonregular"] += 1
                        elif opened.st_dev != device:
                            counts["skipped_other_device"] += 1
                        elif opened.st_nlink != 1:
                            # Skip all aliases; this also excludes external hard links.
                            counts["skipped_hardlink"] += 1
                        else:
                            os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
                            counts["advised"] += 1
                            counts["advised_file_bytes"] += opened.st_size
                    finally:
                        os.close(fd)
                except OSError as error:
                    counts["error_count"] += 1
                    if len(counts["errors"]) < 20:
                        counts["errors"].append({"file": relative + "/" + name,
                                                 "errno": error.errno, "error": str(error)})
    finally:
        os.close(root_fd)
    return counts


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--config", required=True)
    parser.add_argument("--container-id", required=True)
    parser.add_argument("--output", required=True)
    parser.add_argument("--observe-only", action="store_true")
    args = parser.parse_args()
    if sys.platform != "linux" or not re.fullmatch(r"[0-9a-f]{64}", args.container_id):
        raise RuntimeError("Linux and an exact full private container ID are required")
    config = json.loads(Path(args.config).read_text())
    expected = config["containers"].get(args.container_id)
    if expected is None:
        raise RuntimeError("The container is not in the explicit approval manifest")
    output = Path(args.output)
    output_root = Path(config["output_root"]).resolve(strict=True)
    if output.parent.resolve(strict=True) != output_root or output.exists():
        raise RuntimeError("The report must be a new file in the approved output directory")
    identity = inspect_container(args.container_id, expected)
    report = {"started_at": now(), "source_sha": config["source_sha"],
              "helper_sha256": digest(__file__), "approval_sha256": digest(args.config),
              "identity": identity, "before": resources(identity), "roots": [],
              "operation": "observe_only" if args.observe_only else "POSIX_FADV_DONTNEED"}
    if not args.observe_only:
        report["roots"] = [advise_root(root) for root in expected["roots"]]
    after_identity = inspect_container(args.container_id, expected)
    if after_identity != identity:
        raise RuntimeError("The private container identity changed during observation")
    report["after"] = resources(identity)
    report["ended_at"] = now()
    report["error_count"] = sum(root["error_count"] for root in report["roots"])
    report["hint_is_not_reclaimed_bytes"] = True
    with output.open("x") as handle:
        json.dump(report, handle, indent=2)
        handle.write("\n")
    print(json.dumps({"report": str(output), "errors": report["error_count"],
                      "before_current": report["before"]["memory.current"],
                      "after_current": report["after"]["memory.current"]}))
    return 1 if report["error_count"] else 0


if __name__ == "__main__":
    raise SystemExit(main())
